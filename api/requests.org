* App API                                                         :verb:
Template http://localhost:8000/app
# Access-Token: Bearer TOKEN

** Health check                                                       :verb:
GET /health
# Returns: 200 OK with {"status": "healthy"} when healthy
# Returns: 503 Service Unavailable with {"status": "unhealthy"} when unhealthy

** Sign up (create org or join via invite)
POST /auth/sign-up
Content-Type: application/json
# Returns: 201 Created with access token, user, org, and membership

{
  "first_name": "Jane",
  "last_name": "Doe",
  "email": "jane@example.com",
  "password": "Password123!",
  "org_name": "Engineering Team"
}

** Preview invite link
GET /invites/INVITE_TOKEN
# Returns: 200 OK with invite public metadata (org name, role, inviter)

** Authenticate
POST /auth
Content-Type: application/json
# Returns: 200 OK with access token

{
  "email": "user@example.com",
  "password": "11111111"
}

** Logout
POST /logout
# Returns: 200 OK with {"message": "Successfully logged out"}

** Get current user
GET /users/self
# Returns: 200 OK with user details
# Requires: Authentication

** Get current organization
GET /org
# Returns: 200 OK with organization details
# Requires: CurrentAuth

** Update organization
PATCH /org
Content-Type: application/json
# Returns: 200 OK with updated organization
# Requires: CurrentAuth (owner/admin)

{
  "name": "Updated Org Name"
}

** List organization members
GET /org/members
# Returns: 200 OK with list of members and users
# Requires: CurrentAuth

** Update member role
PATCH /org/members/{{(verb-var membership-id)}}
Content-Type: application/json
# Returns: 200 OK with updated member
# Requires: CurrentAuth (owner/admin permissions enforced)

{
  "org_role": "admin"
}

** Remove member
DELETE /org/members/{{(verb-var membership-id)}}
# Returns: 200 OK
# Requires: CurrentAuth (owner/admin, cannot remove last owner)

** List invites
GET /org/invites
# Returns: 200 OK with active invites
# Requires: CurrentAuth (owner/admin)

** Create invite
POST /org/invites
Content-Type: application/json
# Returns: 201 Created with raw_token
# Requires: CurrentAuth (owner/admin)

{
  "org_role": "member",
  "expires_in_days": 7
}

** Revoke invite
DELETE /org/invites/{{(verb-var invite-id)}}
# Returns: 200 OK
# Requires: CurrentAuth (owner/admin)

* Internal API                                                            :verb:
Template http://localhost:8000/internal
# Access-Token: Bearer TOKEN

** Health check                                                       :verb:
GET /health
# Returns: 200 OK with {"status": "healthy"} when healthy
# Returns: 503 Service Unavailable with {"status": "unhealthy"} when unhealthy

** Authenticate
POST /auth
Content-Type: application/json
# Returns: 200 OK with access token
# Note: only users with is_admin may authenticate here

{
  "email": "admin@example.com",
  "password": "adminpass1"
}

** Logout
POST /logout
# Returns: 200 OK with {"message": "Successfully logged out"}

** Get current internal user
GET /users/self
# Returns: 200 OK with internal user details
# Requires: Internal authentication (403 ADMIN_REQUIRED otherwise)

** List users
GET /users?limit=50&offset=0
# Returns: 200 OK with {items, total, limit, offset}
# Query: limit (1-100, default 50), offset (>=0, default 0)
# Requires: Internal authentication

** Create user
POST /users
Content-Type: application/json
# Returns: 201 Created with the new user
# Returns: 409 Conflict if the email already exists
# Requires: Internal authentication

{
  "first_name": "Jane",
  "last_name": "Doe",
  "email": "jane@example.com",
  "password": "secret123",
  "is_admin": false,
  "is_active": true
}

** Update user (partial)
PATCH /users/{{(verb-var user-id)}}
Content-Type: application/json
# Returns: 200 OK with the updated user
# Note: only sent fields change; password is ignored here (use change-password)
# Returns: 404 if not found, 409 on email conflict
# Requires: Internal authentication

{
  "last_name": "Smith",
  "is_admin": true
}

** Change user password
POST /users/{{(verb-var user-id)}}/change-password
Content-Type: application/json
# Returns: 204 No Content
# Requires: Internal authentication

{
  "password": "brandnew1"
}

** Delete user
DELETE /users/{{(verb-var user-id)}}
# Returns: 204 No Content (row hard-deleted)
# Returns: 404 if not found
# Requires: Internal authentication
